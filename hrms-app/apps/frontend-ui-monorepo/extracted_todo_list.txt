2.1 — Create profiles table in Supabase
CREATE TABLE profiles (
  id UUID PRIMARY KEYREFERENCES auth.users(id),
  full_name TEXT,
  role TEXT CHECK (role IN ('hr', 'manager', 'employee')),
  must_change_password BOOLEANDEFAULTtrue,
  avatar_url TEXT,
  created_at TIMESTAMPDEFAULT NOW()
);
2.2 — Create Supabase Edge Function: create-employee-user
Uses Service Role key (never on frontend)
Accepts: email, full_name, temp_password, role
Calls supabase.auth.admin.createUser()
Sets email_confirmed: true
Inserts row into profiles with must_change_password: true
